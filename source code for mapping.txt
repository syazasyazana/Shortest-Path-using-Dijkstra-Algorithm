% Create a graph object
G = graph(s, t, weight);

% Plot the graph with node coordinates
figure;
hold on;

% Plot nodes at specified coordinates
for i = 1:size(coords, 1)
    plot(coords(i, 1), coords(i, 2), 'ro', 'MarkerSize', 10, 'LineWidth', 2);
    text(coords(i, 1), coords(i, 2), sprintf(' %d', i), 'FontSize', 12);
end

% Plot edges with weights
for i = 1:numel(s)
    x = [coords(s(i), 1), coords(t(i), 1)];
    y = [coords(s(i), 2), coords(t(i), 2)];
    plot(x, y, 'b-', 'LineWidth', 2);
    % Calculate the midpoint of the edge for labeling
    midX = mean(x);
    midY = mean(y);
    text(midX, midY, sprintf(' %d', weight(i)), 'FontSize', 10, 'Color', 'b');
end
